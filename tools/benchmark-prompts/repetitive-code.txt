These are two functions from a small settings parser:

def parse_integer(settings, key, default=None):
    raw = settings.get(key)
    if raw is None:
        return default
    if not isinstance(raw, str):
        raise ValueError(f"{key}: expected a string")
    value = raw.strip()
    if not value:
        return default
    try:
        return int(value)
    except ValueError as exc:
        raise ValueError(f"{key}: invalid integer") from exc

def parse_decimal(settings, key, default=None):
    raw = settings.get(key)
    if raw is None:
        return default
    if not isinstance(raw, str):
        raise ValueError(f"{key}: expected a string")
    value = raw.strip()
    if not value:
        return default
    try:
        return float(value)
    except ValueError as exc:
        raise ValueError(f"{key}: invalid decimal") from exc
